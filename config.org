This file is the single source of truth for the Emacs configuration.  On
startup, =init.el= tangles it to =config.el= with Org Babel and loads the
result.  =early-init.el= stays a plain Emacs Lisp file because it runs before
Org and =org-babel-load-file= are available.

* Bootstrap & Early Init

=early-init.el= runs before the package system and before the first frame is
created.  It disables the built-in package manager (Elpaca takes over later),
removes the menu bar, tool bar, and scroll bars, starts fullscreen, and maps
the macOS Command key to Meta while leaving Option free.  It cannot be tangled
from this file, so it remains a separate =early-init.el=.

The bootstrap below is the part of the old =init.el= that must run before any
module: backup and auto-save locations, Elpaca itself, =use-package=
integration, and basic editor defaults.  The module =require= calls are gone
because every module now lives in this document.

* File Backups & Auto-save

Emacs normally writes =foo~= next to the edited file and lock files in the
working directory.  Redirecting backups and auto-saves to =tmp/= keeps project
trees and Git status clean.  Lock files are disabled completely.

#+begin_src emacs-lisp :comments no
;;; config.el --- Tangled from config.org -*- lexical-binding: t; -*-
#+end_src

#+begin_src emacs-lisp

#+end_src

* Package Management (Elpaca)

Elpaca is bootstrapped from Git on first start and then installs packages
asynchronously.  =elpaca-use-package= makes every =use-package= form install
through Elpaca by default, so most forms in this file do not need =:ensure t=.
The =compat= package is pinned to GNU-devel because newer packages expect it.

#+begin_src emacs-lisp
(defvar elpaca-installer-version 0.11)
(defvar elpaca-directory (expand-file-name "elpaca/" user-emacs-directory))
(defvar elpaca-builds-directory (expand-file-name "builds/" elpaca-directory))
(defvar elpaca-repos-directory (expand-file-name "repos/" elpaca-directory))
(defvar elpaca-order '(elpaca :repo "https://github.com/progfolio/elpaca.git"
                              :ref nil
                              :files (:defaults "elpaca-test.el" (:exclude "extensions"))
                              :build (:not elpaca--activate-package)))
(let* ((repo  (expand-file-name "elpaca/" elpaca-repos-directory))
       (build (expand-file-name "elpaca/" elpaca-builds-directory))
       (order (cdr elpaca-order))
       (default-directory repo))
  (add-to-list 'load-path (if (file-exists-p build) build repo))
  (unless (file-exists-p repo)
    (make-directory repo t)
    (when (< emacs-major-version 28) (require 'subr-x))
    (condition-case-unless-debug err
        (if-let ((buffer (pop-to-buffer-same-window "*elpaca-bootstrap*"))
                 ((zerop (call-process "git" nil buffer t "clone"
                                       (plist-get order :repo) repo)))
                 ((zerop (call-process "git" nil buffer t "checkout"
                                       (or (plist-get order :ref) "--"))))
                 (emacs (concat invocation-directory invocation-name))
                 ((zerop (call-process emacs nil buffer nil "-Q" "-L" "." "-batch"
                                       "--eval" "(byte-recompile-directory \".\" 0 'force)")))
                 ((require 'elpaca))
                 ((elpaca-generate-autoloads "elpaca" repo)))
            (progn (message "%s" (buffer-string)) (kill-buffer buffer))
          (error "%s" (funcall #'elpaca--format-error err)))
      ((error) (warn "%s" err) (delete-directory elpaca-directory 'recursive))))
  (unless (require 'elpaca-autoloads nil t)
    (require 'elpaca)
    (elpaca-generate-autoloads "elpaca" repo)
    (load "./elpaca-autoloads")))
(add-hook 'after-init-hook #'elpaca-process-queues)
(elpaca `(,@elpaca-order))

(elpaca elpaca-use-package
  (elpaca-use-package-mode)
  (setq elpaca-use-package-by-default t))
(use-package compat
  :ensure (:source "gnu-devel" :package "compat"))

* Meow & Keybindings

Meow provides modal editing with a selection-first model: select text with a
motion or a "thing" command, then act on the selection.  Four states matter
here.  Normal state holds the editing commands, insert state types text,
motion state is used in special buffers (Dired, Agenda, PDF, Magit, ...) where
the mode's own keys stay available, and keypad state replaces =C-x= and =M-x=
chords.  The layout is QWERTY.

meow-leader-define-key

=SPC= is the leader key in normal and motion state.  =SPC= followed by a key
sequence runs the custom bindings defined with =meow-leader-define-key= (the
same combinations the old leader map used), and =SPC SPC= opens
=meow-keypad=.  Inside the keypad Meow normally treats =SPC= as its "literal"
prefix, so that role is moved to backslash to free =SPC= for this binding.

Motion state binds =j= and =k= to =my/meow-motion-next= and
=my/meow-motion-prev=.  They behave like =meow-next= and =meow-prev= everywhere
except in the modes listed in =my/meow-motion-line-commands= (Agenda, calfw,
PDF), where they call that mode's own item or page commands.  The remaining
mode-specific keys (Dired, Agenda, calfw) are defined in their own sections.
Meow has no visual state, so there is nothing to map for it; selections are
made with the normal-state commands instead.

    Each entry is (MODE NEXT-COMMAND PREVIOUS-COMMAND).  Buffers whose major mode
    is not listed fall back to `meow-next' and `meow-prev'.")

(defun my/display-buffer-no-split (orig buffer-or-name &optional action frame)
  "Force display-buffer to reuse current window unless whitelisted.
Magit and transient buffers are exempt — they manage their own
display action (side window at the bottom)."
  (let* ((name (if (bufferp buffer-or-name)
                   (buffer-name buffer-or-name)
                 (format "%s" buffer-or-name)))
         (magit-cmd
          (and (boundp 'real-this-command)
               (symbolp real-this-command)
               (string-match-p "\\`\\(magit\\|transient\\)"
                               (symbol-name real-this-command))))
         (whitelisted
          (or magit-cmd
              (seq-some (lambda (re) (string-match-p re name))
                        my/popup-whitelist))))
    (funcall orig buffer-or-name
             (if whitelisted
                 action
               '(display-buffer-reuse-window display-buffer-same-window))
             frame)))

** Smart RET and Checkboxes

=my/org-ret-dwim= makes RET context-sensitive: it toggles checkboxes, follows

#+begin_src emacs-lisp
  (defun my/org-toggle-checkbox-3way ()
    "Cycle checkbox: [ ] -> [X] -> [-] -> [ ]."
    (interactive)
    (let ((state (org-element-property :checkbox (org-element-context))))
      (pcase state
        ('off   (org-toggle-checkbox))        ; [ ] -> [X]
        ('on    (org-toggle-checkbox '(16)))  ; [X] -> [-]
        ('trans                               ; [-] -> [ ]
         (org-toggle-checkbox '(4))           ; remove
         (org-toggle-checkbox '(4))))))       ; add again

  (defun my/org-ret-dwim ()
    "Smart RET in normal state: links, tables, lists, headings."
    (interactive)
    (let* ((element (org-element-context))
           (item    (org-element-lineage element '(item) t)))
      (cond
       ((eq major-mode 'sqlite-mode)
        (call-interactively 'sqlite-mode-list-data))
       ((eq 'link (org-element-type element))
        (org-open-at-point-global))
       ((org-at-table-p)

       (t (call-interactively #'meow-next)))))

** Moving DONE Tasks (plans.org)

The autosort function moves a task to the end of the =Уже сделано= or =:done:=
section after it is marked DONE.  It runs 0.1 seconds after the state change
so that Org has finished updating the buffer.

#+begin_src emacs-lisp
;; ================= plans.org autosort ================
;; ====== Move DONE tasks to the "Уже сделано" section ======
(defun my/org-move-done-to-bottom ()
  "Move a task to the end of the '* Уже сделано :done:' section when marked DONE."
  (when (equal org-state "DONE")
    (let ((task-marker (point-marker)))
      (run-at-time 0.1 nil
                   (lambda (marker)
                     (when (marker-buffer marker)
                       (with-current-buffer (marker-buffer marker)
                         (save-excursion
                           (goto-char marker)
                           (when (org-at-heading-p)
                             (org-back-to-heading t)
                             ;; Check whether we are already inside that section
                             (let ((parent-heading
                                    (save-excursion
                                      (ignore-errors
                                        (org-up-heading-safe)
                                        (org-get-heading t t t t)))))
                               (unless (and parent-heading
                                            (string-match-p "Уже сделано" parent-heading))

                                 ;; Find the section by :done: tag or by name
                                 (let ((target-marker
                                        (save-excursion
                                          (goto-char (point-min))
                                          (when (or (re-search-forward "^\\*+ .*:done:" nil t)
                                                    (re-search-forward "^\\*+ Уже сделано" nil t))
                                            ;; The section is always at the end, so jump to the end of the buffer
                                            (goto-char (point-max))
                                            (copy-marker (point))))))

                                   (if (marker-position target-marker)
                                       (progn
                                         ;; 1. Cut the task
                                         (org-cut-subtree)
                                         ;; 2. Go to the end of the file and paste at level 2 (**)
                                         (goto-char target-marker)
                                         (unless (bolp) (newline))
                                         (org-paste-subtree 2)
                                         (set-marker target-marker nil)

                                         ;; 3. Fold the "Уже сделано" section
                                         (save-excursion
                                           (goto-char (point-min))
                                           (when (or (re-search-forward "^\\*+ .*:done:" nil t)
                                                     (re-search-forward "^\\*+ Уже сделано" nil t))
                                             (org-back-to-heading t)
                                             (org-fold-hide-subtree)))

                                         (message "Task moved to 'Уже сделано' and section folded."))
                                     (message "Warning: Section with tag ':done:' or name 'Уже сделано' not found.")))))))))
                     (set-marker marker nil))
                   task-marker))))

* Russian Keyboard Layout (reverse-im)

=reverse-im= translates input typed in the Russian layout into the Latin keys
that Emacs commands expect.  This keeps leader and other keybindings usable
while the system layout is Russian.

#+begin_src emacs-lisp
(use-package reverse-im
  :ensure t
  :after meow
  :custom
      (reverse-im-input-methods '("russian-computer"))
  :config
    (reverse-im-mode t))
;; reverse-im only translates key bindings.  Prompts that read a raw character
;; (Meow's "thing" prompt, org-agenda and capture menus) still see Cyrillic.
;; Translate those reads, but only inside the commands listed below, so that
;; `f' / `t' can still search for Cyrillic characters.
(defconst my/ru-en-table
  (let ((h (make-hash-table)))
    (cl-loop for r across "йцукенгшщзхъфывапролджэячсмитьбюё"
             for e across "qwertyuiop[]asdfghjkl;'zxcvbnm,.`"
             do (puthash r e h))
    (cl-loop for r across "ЙЦУКЕНГШЩЗХЪФЫВАПРОЛДЖЭЯЧСМИТЬБЮЁ"
             for e across "QWERTYUIOP{}ASDFGHJKL:\"ZXCVBNM<>~"
             do (puthash r e h))
    h)
  "Russian (ЙЦУКЕН) to QWERTY character map.")

(defvar my/ru-translate-reads nil
  "When non-nil, raw character reads are translated from Russian to QWERTY.")

(defun my/ru-translate-read-a (result)
  (if (and my/ru-translate-reads (integerp result))
      (gethash result my/ru-en-table result)
    result))

(dolist (fn '(read-char read-char-exclusive read-event read-key))
  (advice-add fn :filter-return #'my/ru-translate-read-a))

  (add-hook 'after-save-hook #'my/typst--on-save nil t)
  (my/typst--bind-keys))

(with-eval-after-load 'lsp-mode
  ;; Регистрируем language-id для всех субмодов typst
  (dolist (mode '(typst-mode
                  typst--markup-mode

(defvar my/typst--watch-process nil)
(defvar my/typst--preview-buffer nil)
(defvar my/typst--refresh-timer nil)

(defun my/typst--pdf-file ()
  (concat (file-name-sans-extension (buffer-file-name)) ".pdf"))

(defun my/typst--watch-alive-p ()
  (and my/typst--watch-process (process-live-p my/typst--watch-process)))

(defun my/typst--stop-watch ()
  (when (my/typst--watch-alive-p)
    (delete-process my/typst--watch-process))
  (setq my/typst--watch-process nil))

(defun my/typst--pdf-buffer ()
  (when (buffer-live-p my/typst--preview-buffer)
    my/typst--preview-buffer))

(defun my/typst--refresh-pdf (&rest _)
  "Revert preview PDF buffer in place, безопасно к гонкам."
  (let ((buf (my/typst--pdf-buffer)))
    (when buf
      (with-current-buffer buf
        (when (and (derived-mode-p 'pdf-view-mode)
                   (file-exists-p (buffer-file-name)))
          (condition-case err
              (pdf-view-revert-buffer nil t)
            (error (message "PDF refresh failed: %s" err))))))))

(defun my/typst--schedule-refresh ()
  "Debounce-перерисовка PDF: watch печатает несколько строк подряд."
  (when my/typst--refresh-timer
    (cancel-timer my/typst--refresh-timer))
  (setq my/typst--refresh-timer
        (run-at-time 0.2 nil #'my/typst--refresh-pdf)))

(defun my/typst--compile-sync ()
  "Синхронная компиляция в PDF. Возвращает t при успехе."
  (zerop (call-process typst-executable-location
                       nil "*typst-compile*" nil
                       "compile"
                       (buffer-file-name)
                       (my/typst--pdf-file))))

(defun my/typst--compile-async ()
  "Compile current file to PDF in background (fallback без watch)."
  (when (buffer-file-name)
    (let ((proc (start-process
                 "typst-compile" "*typst-compile*"
                 typst-executable-location "compile"
                 (buffer-file-name) (my/typst--pdf-file))))
      (set-process-sentinel
       proc
       (lambda (_p _e) (my/typst--refresh-pdf))))))

(defun my/typst--on-save ()
  "Вызывается при сохранении .typ."
  (when (my/typst--pdf-buffer)
    (if (my/typst--watch-alive-p)
        ;; watch сам перекомпилирует → process-filter вызовет refresh.
        nil
      (my/typst--compile-async))))

(defun my/typst--open-pdf-window ()
  "Open PDF next to source, remember buffer."
  (let* ((pdf (my/typst--pdf-file))
         (buf nil))
    (unless (file-exists-p pdf)
      (user-error "PDF не появился: %s" pdf))
    (setq buf (find-file-noselect pdf))
    (setq my/typst--preview-buffer buf)
    (if (> (window-total-width) 140)
        (display-buffer-in-side-window
         buf
         '((side . right)
           (window-width . 0.45)
           (dedicated . t)))
      (display-buffer
       buf
       '((display-buffer-reuse-window display-buffer-pop-up-window)
         (window-width . 0.5))))
    (with-current-buffer buf
      (pdf-view-mode)
      (pdf-view-fit-page-to-window)
      (display-line-numbers-mode -1)
      (setq-local mode-line-format
                  (list " " mode-line-buffer-identification "  [typst preview] ")))
    buf))

(defun my/typst-live-preview ()
  "Start live preview: typst watch + PDF в соседнем окне."
  (interactive)
  (unless (buffer-file-name)
    (user-error "Буфер не сохранён в файл"))
  (unless (executable-find typst-executable-location)
    (user-error "typst не найден в PATH"))
  (save-buffer)

  ;; Первая компиляция — синхронно, чтобы PDF был валиден к моменту открытия.
  (unless (my/typst--compile-sync)
    (user-error "typst compile упал, смотри *typst-compile*"))

  (my/typst--stop-watch)
  (let* ((input (file-name-nondirectory (buffer-file-name)))
         (output (concat (file-name-sans-extension input) ".pdf"))
         (default-directory (file-name-directory (buffer-file-name))))
    (setq my/typst--watch-process
          (make-process
           :name "typst-watch"
           :buffer "*typst-watch*"
           :command (list typst-executable-location "watch" input output)
           ;; watch печатает в stdout после каждой перекомпиляции.
           :filter (lambda (_proc _output)
                     (my/typst--schedule-refresh))
           :sentinel
           (lambda (p _e)
             (unless (process-live-p p)
               (message "typst watch остановлен")))))
    (message "typst watch запущен"))
  (my/typst--open-pdf-window))

(defun my/typst-stop-preview ()
  "Stop watch and close preview PDF."
  (interactive)
  (my/typst--stop-watch)
  (when my/typst--refresh-timer
    (cancel-timer my/typst--refresh-timer)
    (setq my/typst--refresh-timer nil))
  (let ((buf (my/typst--pdf-buffer)))
    (when buf
      (setq my/typst--preview-buffer nil)
      (when-let ((win (get-buffer-window buf)))
        (delete-window win))
      (kill-buffer buf)))
  (message "Typst preview остановлен"))

(defun my/typst-compile ()
  "Compile current file to PDF once (synchronously)."
  (interactive)
  (unless (buffer-file-name)
    (user-error "Буфер не сохранён в файл"))
  (save-buffer)
  (if (my/typst--compile-sync)
      (message "Скомпилировано: %s" (my/typst--pdf-file))
    (message "Ошибка компиляции — смотри *typst-compile*")))

(defun my/typst--bind-keys ()
  "C-c bindings for typst buffers.
Meow's normal state does not bind control-key chords, so these work in every
Meow state."
  (when (boundp 'typst-mode-map)
    (define-key typst-mode-map (kbd "C-c C-c") #'my/typst-compile)
    (define-key typst-mode-map (kbd "C-c C-p") #'my/typst-live-preview)
    (define-key typst-mode-map (kbd "C-c C-s") #'my/typst-stop-preview)
    (define-key typst-mode-map (kbd "C-c C-r") #'my/typst--refresh-pdf)))

(defun my/typst-stop-or-magit-status ()
  "SPC m s: stop the Typst preview in Typst buffers, `magit-status' elsewhere."
  (interactive)
  (if (string-prefix-p "typst" (symbol-name major-mode))
      (call-interactively #'my/typst-stop-preview)
    (call-interactively #'magit-status)))

(with-eval-after-load 'meow

   '("m c" . my/typst-compile)
   '("m p" . my/typst-live-preview)
   '("m s" . my/typst-stop-or-magit-status)
   '("m r" . my/typst--refresh-pdf)))

* Dired & File Management

Dired is replaced by =dirvish=, which adds icons, Git state, file sizes and
times, a header with the path and free space, and quick sorting.  Meow motion-state keys
navigate the tree and manage files, while =dired-ranger= provides copy, move,
and paste.  Deletion confirmations are disabled, recursive operations run
silently, and deleted files go to the Trash.

* Version Control (Magit)

Magit is bound to =C-x g= for status and =C-x M-g= for dispatch.  A custom
display function keeps diff, log, and revision buffers in a bottom window,
while all other Magit buffers reuse the selected window.  A matching advice
deletes that popup window when Magit is buried.

  (defun my/magit-display-buffer (buffer)
    "Display BUFFER for Magit.
  Diff/log/revision buffers open in a window at the bottom of the
  selected frame; everything else opens in the selected window.
  Respects `magit-display-buffer-noselect' so the staged diff shown
  next to the commit message does not steal focus."
    (if (with-current-buffer buffer
          (derived-mode-p 'magit-diff-mode
                          'magit-revision-mode
                          'magit-log-mode))
        (let ((win (get-buffer-window buffer)))
          (unless (window-live-p win)
            (let* ((total (window-height (selected-window)))
                   (size  (max 10 (round (* 0.4 total)))))
              (setq win (split-window-below size))
              (set-window-parameter win 'my/magit-popup t)))
          (set-window-buffer win buffer)
          ;; NEW: не забираем фокус, если Magit сказал "не надо"
          (unless magit-display-buffer-noselect
            (select-window win))
          win)
      (display-buffer buffer '(display-buffer-same-window))))

    "Run ORIG (bury), then delete the popup window we created for it."
    (let ((win (selected-window)))
      (prog1 (apply orig args)
        (when (and (window-live-p win)
                   (window-parameter win 'my/magit-popup)
                   (not (one-window-p)))
          (delete-window win)))))

           ("C-x M-g" . magit-dispatch))
    :config

* Custom Utilities & Global Modes

This section holds personal helpers and global modes that do not belong to a
package.  Auto-revert refreshes buffers silently when files change and
electric-pair closes brackets.  =add-songs= collects track lines into a
=Tracks= section, and two commands kill Org Roam or Magit buffers in bulk.

#+begin_src emacs-lisp
(require 'org)

(defun add-songs ()
  "Собрать все строки треков (- ...) из текущего Org-файла
и поместить их единым списком под заголовком * Tracks в конце буфера."
  (interactive)
  (save-excursion
        (goto-char (point-min))
    (when (re-search-forward "^\* Tracks\\(\\s-\\|$\\)" nil t)
      (beginning-of-line)
      (org-mark-subtree)
      (delete-region (region-beginning) (region-end))
      (message "Старая секция Tracks удалена."))
        (let (tracks)
      (goto-char (point-min))
      (while (re-search-forward "^\\([ \t]*\\)- \\(.*\\)" nil t)
        (push (match-string 2) tracks))
      (setq tracks (nreverse tracks))
            (goto-char (point-max))
            (unless (bolp) (insert "\n"))
      (unless (looking-back "\n\n" nil)
        (insert "\n"))
      (insert "* Tracks\n")
      (dolist (track tracks)
        (insert "- " track "\n"))
      (message "Добавлено %d треков в * Tracks." (length tracks)))))
(global-auto-revert-mode 1)
(setq auto-revert-verbose nil)   ; silence "Reverting buffer ..." messages
(electric-pair-mode t)

(defun my/kill-org-roam-buffers ()
  "Kill all Org-roam buffers (files under `org-roam-directory').
Does not kill the org-roam side buffer (`*org-roam*')."
  (interactive)
  (let ((roam-dir (file-truename org-roam-directory))
        (count 0))
    (dolist (buf (buffer-list))
      (with-current-buffer buf
        (when (and buffer-file-name
                   (string-prefix-p roam-dir (file-truename buffer-file-name)))
          (kill-buffer buf)
          (setq count (1+ count)))))
    (message "Killed %d org-roam buffer(s)" count)))

(defun my/kill-all-magit-buffers ()
  "Kill all Magit-related buffers (status, diffs, logs, processes, etc.)."
  (interactive)
  (let ((count 0))
    (dolist (buf (buffer-list))
      (with-current-buffer buf
        ;; Проверяем, что это Magit-буфер либо по имени, либо по major-mode
        (when (or (string-prefix-p "*magit" (buffer-name buf))
                  (and major-mode
                       (string-prefix-p "magit-" (symbol-name major-mode))))
          (kill-buffer buf)
          (setq count (1+ count)))))
    (message "Killed %d Magit buffer(s)" count)))
